# The Ribbit platform: vendored in third_party/ribbit; -DRIBBIT_ROOT=<a FrogNet tree's ribbit directory> to use that instead
set(RIBBIT_ROOT ${CMAKE_SOURCE_DIR}/third_party/ribbit CACHE PATH "the Ribbit platform: vendored by default, or a FrogNet tree's ribbit directory")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Compiler flags: GCC/Clang (Linux, MSYS2) or MSVC (Visual Studio). Warnings are always shown. They stop the build only in development (-DCOMMS_WERROR=ON): every GCC release reports
# its own false positives inside the standard library (GCC 11 stringop-overread, GCC 12 array-bounds), and a build on
# someone else's machine must not fail on a diagnostic.
option(COMMS_WERROR "treat warnings as errors (development)" OFF)
# FFmpeg 4.4 (Ubuntu 22.04) refuses to be included from C++ without this; harmless with later versions.
add_compile_definitions(__STDC_CONSTANT_MACROS)
if(MSVC)
  # /utf-8: the window's strings are UTF-8. _USE_MATH_DEFINES: M_PI. NOMINMAX: windows.h must not define min/max.
  add_compile_options(/W3 /utf-8 /EHsc /permissive- /Zc:__cplusplus)
  add_compile_definitions(_USE_MATH_DEFINES NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS _WINSOCK_DEPRECATED_NO_WARNINGS)
  if(COMMS_WERROR)
    add_compile_options(/WX)
  endif()
else()
  add_compile_options(-Wall -Wextra)
  if(WIN32)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
  endif()
  if(COMMS_WERROR)
    add_compile_options(-Werror)
  endif()
endif()
set(QUIET_VENDORED $<IF:$<CXX_COMPILER_ID:MSVC>,/w,-w>)   # the vendored platform is not ours to warn about
# Libraries the platform links, found the same way on every toolchain
find_package(Threads REQUIRED)
# Libraries, behind one name each (dep_*). Visual Studio + vcpkg: vcpkg's CMake packages -- no pkg-config tool needed
# on Windows (vcpkg's pkgconf port failed to build there, 2026-10-01). Linux and MSYS2: pkg-config.
foreach(d dep_av dep_opus dep_pa dep_lz4 dep_crypto)
  add_library(${d} INTERFACE)
endforeach()
if(MSVC)
  find_package(FFMPEG REQUIRED)                          # vcpkg's FindFFMPEG: every component it built
  target_include_directories(dep_av INTERFACE ${FFMPEG_INCLUDE_DIRS})
  target_link_directories(dep_av INTERFACE ${FFMPEG_LIBRARY_DIRS})
  target_link_libraries(dep_av INTERFACE ${FFMPEG_LIBRARIES})
  find_package(Opus CONFIG REQUIRED)
  target_link_libraries(dep_opus INTERFACE Opus::opus)
  find_package(portaudio CONFIG REQUIRED)                # the target's name changed between releases
  if(TARGET PortAudio::portaudio)
    target_link_libraries(dep_pa INTERFACE PortAudio::portaudio)
  elseif(TARGET portaudio)
    target_link_libraries(dep_pa INTERFACE portaudio)
  else()
    target_link_libraries(dep_pa INTERFACE portaudio_static)
  endif()
  find_package(lz4 CONFIG REQUIRED)
  target_link_libraries(dep_lz4 INTERFACE lz4::lz4)
  find_package(OpenSSL REQUIRED)
  target_link_libraries(dep_crypto INTERFACE OpenSSL::Crypto)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(AVALL REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale libavdevice libavformat)
  pkg_check_modules(OPUS REQUIRED IMPORTED_TARGET opus)
  pkg_check_modules(PA REQUIRED IMPORTED_TARGET portaudio-2.0)
  pkg_check_modules(LZ4 REQUIRED IMPORTED_TARGET liblz4)
  pkg_check_modules(CRYPTO REQUIRED IMPORTED_TARGET libcrypto)
  target_link_libraries(dep_av INTERFACE PkgConfig::AVALL)
  target_link_libraries(dep_opus INTERFACE PkgConfig::OPUS)
  target_link_libraries(dep_pa INTERFACE PkgConfig::PA)
  target_link_libraries(dep_lz4 INTERFACE PkgConfig::LZ4)
  target_link_libraries(dep_crypto INTERFACE PkgConfig::CRYPTO)
endif()
